# 标准化数据加载模块：本地数据读取与预处理代码
# 说明：执行此代码块将自动完成数据集读取、矩阵展平与归一化清洗，为后续算法提供标准的输入数据。
# 注意：请确保已解压的 `data` 文件夹与当前运行的 Jupyter Notebook 处于同一目录下。
# ==========================================
import os
import gzip
import numpy as np

print("正在读取本地数据文件...")

# 1. 定义读取本地离线 gzip 数据的函数 (完全无需联网)
def load_mnist(path, kind='train'):
    labels_path = os.path.join(path, f'{kind}-labels-idx1-ubyte.gz')
    images_path = os.path.join(path, f'{kind}-images-idx3-ubyte.gz')
    
    with gzip.open(labels_path, 'rb') as lbpath:
        labels = np.frombuffer(lbpath.read(), dtype=np.uint8, offset=8)
        
    with gzip.open(images_path, 'rb') as imgpath:
        # 直接在读取时展平为 784
        images = np.frombuffer(imgpath.read(), dtype=np.uint8, offset=16).reshape(len(labels), 784)
        
    return images, labels

# 2. 从当前目录的 data 文件夹加载数据
try:
    data_path = './data'
    X_train_raw, y_train = load_mnist(data_path, kind='train')
    X_test_raw, y_test = load_mnist(data_path, kind='t10k')
except Exception as e:
    print(f"❌ 读取数据失败！请检查：有没有把 'data' 文件夹和这个代码文件放在一起？\n详细错误: {e}")
    raise

# 3. 归一化 (将 0-255 的像素亮度值缩小到 0-1 的区间内，防止梯度爆炸并加速模型收敛)
# 注意：X_train_raw 和 X_test_raw 在上面的函数里已经展平过(784)了
X_train = X_train_raw / 255.0
X_test = X_test_raw / 255.0

print(f"✅ 本地数据读取、清洗与归一化完毕！")
print(f"展平并归一化后的 训练集形状: {X_train.shape}, 测试集形状: {X_test.shape}")
print("请在下方新建代码块，开始编写你负责的算法代码...")

# 随机森林训练模块 Fashion MNIST 10分类
# ==========================================
import time
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import classification_report, accuracy_score

# 类别映射，和截图完全对应
label_names = [
    "T恤/上衣", "裤子", "套头衫", "连衣裙", "外套",
    "凉鞋", "衬衫", "运动鞋", "包", "短靴"
]

start = time.time()
# 随机森林参数，复现你0.83整体准确率结果
rf = RandomForestClassifier(n_estimators=100, random_state=42, n_jobs=-1)
rf.fit(X_train, y_train)
y_pred = rf.predict(X_test)
train_time = time.time() - start

# 打印整体准确率
acc = accuracy_score(y_test, y_pred)
print(f"整体准确率accuracy = {acc:.2f}")
print(f"训练耗时：{train_time:.2f} s")

# 输出完整分类报告（和你第二张截图格式完全一致）
print("\n===== 分类报告 =====")
print(classification_report(y_test, y_pred, target_names=label_names))
